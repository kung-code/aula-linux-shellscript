Shell Script: básico

#!/bin/bash          # shebang
nome="Ana"           # sem espaços no =
echo "Olá, $nome"    # "" expande, '' não
hoje=$(date +%d/%m)  # saída de comando
soma=$((2 + 3))      # aritmética
read -p "Idade? " idade
echo "$1 $# $@ $?"   # args, qtd, todos, status
chmod +x script.sh && ./script.sh arg1
